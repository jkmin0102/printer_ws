cmake_minimum_required(VERSION 3.8)
project(namo_rviz)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(gazebo REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(namo_status_panel SHARED src/status_panel.cpp include/namo_rviz/status_panel.hpp)
target_include_directories(namo_status_panel PUBLIC include)
target_link_libraries(namo_status_panel Qt5::Widgets)
ament_target_dependencies(namo_status_panel rclcpp rviz_common pluginlib std_msgs nav_msgs)
add_library(namo_gazebo_view SHARED src/gazebo_view.cpp)
target_include_directories(namo_gazebo_view SYSTEM PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(namo_gazebo_view PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(namo_gazebo_view Qt5::Widgets gazebo_gui gazebo_common)
target_compile_features(namo_gazebo_view PRIVATE cxx_std_17)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS namo_status_panel namo_gazebo_view LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_package()
